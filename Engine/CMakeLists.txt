# ── Vulkan backend dependencies ──────────────────────────────────────────────
# Fetched (not the system loader): we load entry points via volk with
# VK_NO_PROTOTYPES, so MyEngine never links the Vulkan loader directly. The
# installed Vulkan SDK still supplies validation layers + glslangValidator at
# dev time. See Docs/vulkan-renderer-design.md.
if(DIAMOND_ENABLE_VULKAN)
    FetchContent_Declare(
        vulkan_headers
        GIT_REPOSITORY https://github.com/KhronosGroup/Vulkan-Headers.git
        GIT_TAG        v1.4.350
        GIT_SHALLOW    TRUE
    )
    FetchContent_MakeAvailable(vulkan_headers)

    # Point volk's header search at the fetched Vulkan-Headers (its CMake checks
    # VULKAN_HEADERS_INSTALL_DIR first). Forces use of our pinned 1.4.350 headers
    # rather than whatever the installed SDK happens to be. ON+FORCE also clears
    # any stale VOLK_PULL_IN_VULKAN cache value.
    set(VOLK_PULL_IN_VULKAN ON CACHE BOOL "" FORCE)
    set(VULKAN_HEADERS_INSTALL_DIR "${vulkan_headers_SOURCE_DIR}" CACHE PATH "" FORCE)
    FetchContent_Declare(
        volk
        GIT_REPOSITORY https://github.com/zeux/volk.git
        GIT_TAG        vulkan-sdk-1.4.350.0
        GIT_SHALLOW    TRUE
    )
    FetchContent_MakeAvailable(volk)

    # VMA is header-only; its single implementation TU is compiled into MyEngine
    # via src/Platform/Vulkan/vma_impl.cpp (mirrors the miniaudio_impl.c pattern).
    FetchContent_Declare(
        vma
        GIT_REPOSITORY https://github.com/GPUOpen-LibrariesAndSDKs/VulkanMemoryAllocator.git
        GIT_TAG        v3.4.0
        GIT_SHALLOW    TRUE
    )
    FetchContent_MakeAvailable(vma)
endif()

# ── Vulkan backend wiring ────────────────────────────────────────────────────
# volk + VMA are PRIVATE: Vulkan types never appear in MyEngine's public headers
# (same discipline as Jolt). VK_NO_PROTOTYPES forces all calls through volk's
# loaded pointers. DIAMOND_ENABLE_VULKAN gates the backend at compile time.
if(DIAMOND_ENABLE_VULKAN)
    target_sources(MyEngine PRIVATE
        src/Platform/Vulkan/vma_impl.cpp
        src/Platform/Vulkan/VulkanContext.cpp
        src/Platform/Vulkan/VulkanSwapchain.cpp
        src/Platform/Vulkan/VulkanBuffer.cpp
        src/Platform/Vulkan/VulkanImage.cpp
        src/Platform/Vulkan/RHI/VulkanRHIDevice.cpp
        src/Platform/Vulkan/RHI/VulkanRHIResources.cpp
        src/Platform/Vulkan/Passes/PostProcess/VulkanTonemapPass.cpp
        src/Platform/Vulkan/Passes/PostProcess/VulkanFXAAPass.cpp
        src/Platform/Vulkan/Passes/PostProcess/VulkanBloomPass.cpp
        src/Platform/Vulkan/Passes/Deferred/VulkanGBufferPass.cpp
        src/Platform/Vulkan/Passes/Deferred/VulkanSSAOPass.cpp
        src/Platform/Vulkan/Passes/Deferred/VulkanDeferredLightingPass.cpp
        src/Platform/Vulkan/Passes/Forward/VulkanPBRSurfacePass.cpp
        src/Platform/Vulkan/Passes/Forward/VulkanTransparencyPass.cpp
        src/Platform/Vulkan/Passes/Shadows/VulkanCSMPass.cpp
        src/Platform/Vulkan/Passes/IBL/VulkanIBLPass.cpp
        src/Platform/Vulkan/Resources/VulkanRenderer2D.cpp
        src/Platform/Vulkan/Resources/VulkanTexture2D.cpp
        src/Platform/Vulkan/VulkanImGuiLayer.cpp
        src/Platform/Vulkan/VulkanMeshDemo.cpp
        src/Platform/Vulkan/VulkanSceneDemo.cpp
        # ImGui's Vulkan renderer backend. Compiled into MyEngine (not the imgui
        # static lib) so it inherits VK_NO_PROTOTYPES + the volk include path; USE_VOLK
        # makes it call volk's already-loaded function pointers (no LoadFunctions dance).
        ${imgui_SOURCE_DIR}/backends/imgui_impl_vulkan.cpp
    )
    target_link_libraries(MyEngine PRIVATE
        volk
        VulkanMemoryAllocator
    )

    # ── Compile the Vulkan backend's GLSL → SPIR-V at build time ─────────────
    # glslangValidator ships with the Vulkan SDK. Offline compilation keeps the
    # dependency footprint light (no shaderc/glslang FetchContent); runtime
    # libshaderc for hot-reload arrives with the material/shader-library port. The
    # output dir is baked in as DIAMOND_VULKAN_SHADER_DIR so the loader needs no
    # working-dir assumptions.
    find_program(GLSLANG_VALIDATOR NAMES glslangValidator
        HINTS $ENV{VULKAN_SDK}/Bin $ENV{VULKAN_SDK}/bin)
    if(NOT GLSLANG_VALIDATOR)
        message(FATAL_ERROR
            "glslangValidator not found — install the Vulkan SDK (or set VULKAN_SDK). "
            "Required to compile the Vulkan backend's shaders.")
    endif()

    set(_vk_shader_src ${CMAKE_SOURCE_DIR}/Assets/Shaders/Vulkan)
    set(_vk_spv_dir    ${CMAKE_BINARY_DIR}/shaders)
    file(MAKE_DIRECTORY ${_vk_spv_dir})
    set(_vk_spv_outputs)
    foreach(_shader mesh.vert mesh.frag fullscreen.vert tonemap.frag fxaa.frag
                    bloom_extract.frag bloom_blur.frag bloom_composite.frag
                    gbuffer.vert gbuffer.frag gbuffer_debug.frag
                    ssao.frag ssao_blur.frag
                    csm_depth.vert csm_depth.frag
                    deferred_lighting.frag
                    cubemap.vert equirect_to_cubemap.frag
                    irradiance_convolution.frag prefilter.frag brdf_lut.frag
                    copy.frag skybox.vert skybox.frag
                    pbr_surface.vert pbr_surface.frag
                    transparent.vert transparent.frag
                    renderer2d.vert renderer2d.frag)
        add_custom_command(
            OUTPUT  ${_vk_spv_dir}/${_shader}.spv
            COMMAND ${GLSLANG_VALIDATOR} -V ${_vk_shader_src}/${_shader} -o ${_vk_spv_dir}/${_shader}.spv
            DEPENDS ${_vk_shader_src}/${_shader}
            COMMENT "Compiling Vulkan shader ${_shader} → SPIR-V"
            VERBATIM)
        list(APPEND _vk_spv_outputs ${_vk_spv_dir}/${_shader}.spv)
    endforeach()
    add_custom_target(VulkanShaders DEPENDS ${_vk_spv_outputs})
    add_dependencies(MyEngine VulkanShaders)

    target_compile_definitions(MyEngine PRIVATE
        DIAMOND_ENABLE_VULKAN
        VK_NO_PROTOTYPES
        IMGUI_IMPL_VULKAN_USE_VOLK   # imgui_impl_vulkan calls volk's loaded pointers
        DIAMOND_VULKAN_SHADER_DIR="${_vk_spv_dir}"
        DIAMOND_ASSETS_DIR="${CMAKE_SOURCE_DIR}/Assets"
    )
endif()
